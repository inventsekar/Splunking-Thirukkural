step 4
to find out the character count:

source="numberedKurals-charCount.txt" host="test" sourcetype="kural" 
| rex max_match=0 (?<digits>\d) 
| eval digitsCount=mvcount(digits) 
| rex max_match=0 (?<char>.) 
| lookup ucd_category_lookup char output category 
| eval length=mvcount(mvfilter(NOT match(category, "^M"))) 
| eval len = length - digitsCount 
| eval CharCount=len-1 
| table _raw digits digitsCount length len CharCount
